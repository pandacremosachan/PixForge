cmake_minimum_required(VERSION 3.21)
project(PixForgeNative VERSION 2.9.0 LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "The native Win32 desktop currently supports Windows only. The core CLI/MCP remain available on other platforms.")
endif()
enable_language(RC)
add_executable(PixForgeNative WIN32 main.cpp app.cpp app_scene.cpp scene_editor.cpp minecraft_editor.cpp minecraft_preview.cpp minecraft_workbench.cpp material_editor.cpp app.rc)
target_include_directories(PixForgeNative PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_features(PixForgeNative PRIVATE cxx_std_20)
target_compile_definitions(PixForgeNative PRIVATE UNICODE _UNICODE _WIN32_WINNT=0x0A00 WINVER=0x0A00)
if(MSVC)
  target_compile_options(PixForgeNative PRIVATE /W4 /utf-8 /permissive-)
  set_property(TARGET PixForgeNative PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_link_options(PixForgeNative PRIVATE "/MANIFEST:EMBED" "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/native.manifest")
endif()
target_link_libraries(PixForgeNative PRIVATE user32 gdi32 gdiplus comctl32 comdlg32 shell32 ole32 winhttp ws2_32 bcrypt opengl32 uxtheme dwmapi)
set(PIXFORGE_PYTHON "${CMAKE_CURRENT_SOURCE_DIR}/../.venv/Scripts/python.exe" CACHE FILEPATH "Private application Python for development builds")
file(GENERATE OUTPUT "$<TARGET_FILE_DIR:PixForgeNative>/pixforge-development.json" CONTENT "{\n  \"schema\": \"pixforge.development-install/v1\",\n  \"python\": \"${PIXFORGE_PYTHON}\",\n  \"entry\": \"${CMAKE_CURRENT_SOURCE_DIR}/../app_entry.py\",\n  \"workspace\": \"${CMAKE_CURRENT_SOURCE_DIR}/../../workspace\"\n}\n")
add_custom_command(TARGET PixForgeNative POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/../../public-api.json" "$<TARGET_FILE_DIR:PixForgeNative>/application.json")
